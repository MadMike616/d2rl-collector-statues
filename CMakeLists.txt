cmake_minimum_required(VERSION 3.29)
project(CollectorStatues VERSION 1.0.0 LANGUAGES CXX RC ASM_MASM)
find_package(D2RLPlugin 0.4.0 EXACT CONFIG REQUIRED)
add_library(collector-statues SHARED src/plugin.cpp src/native_tooltip.cpp src/native_sound.cpp src/statue_drops.cpp src/tooltip_thunks.asm src/drop_thunk.asm src/plugin.rc)
target_link_libraries(collector-statues PRIVATE D2RLPlugin::D2RLPlugin user32 bcrypt)
target_compile_features(collector-statues PRIVATE cxx_std_20)
target_compile_definitions(collector-statues PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_compile_options(collector-statues PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/permissive-;/utf-8;/Brepro>")
set_target_properties(collector-statues PROPERTIES PREFIX "d2rl-" MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")

option(COLLECTOR_STATUES_BUILD_TESTS "Build tests using synthetic fixtures" OFF)
if(COLLECTOR_STATUES_BUILD_TESTS)
    enable_testing()
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    add_executable(opening_tests tests/opening_tests.cpp src/native_tooltip.cpp src/native_sound.cpp src/statue_drops.cpp src/tooltip_thunks.asm src/drop_thunk.asm)
    add_executable(drop_tests tests/drop_tests.cpp src/drop_thunk.asm)
    target_link_libraries(opening_tests PRIVATE user32 bcrypt)
    target_link_libraries(drop_tests PRIVATE bcrypt)
    add_executable(bridge_tests tests/bridge_tests.cpp tests/bridge_fixture.asm src/drop_thunk.asm)
    add_executable(catalog_tests tests/catalog_tests.cpp)
    add_executable(sound_tests tests/sound_tests.cpp)
    foreach(test_target IN ITEMS opening_tests drop_tests bridge_tests catalog_tests sound_tests)
        target_link_libraries(${test_target} PRIVATE D2RLPlugin::D2RLPlugin)
        target_compile_features(${test_target} PRIVATE cxx_std_20)
        target_compile_definitions(${test_target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
        target_compile_options(${test_target} PRIVATE "$<$<COMPILE_LANGUAGE:CXX>:/W4;/WX;/permissive-;/utf-8>")
        set_target_properties(${test_target} PROPERTIES MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
    endforeach()
    set(fixture_root "${CMAKE_CURRENT_BINARY_DIR}/test-data")
    set(fixture_files "${fixture_root}/data/global/excel/levels.txt" "${fixture_root}/data/global/excel/monstats.txt" "${fixture_root}/data/global/excel/uniqueitems.txt")
    add_custom_command(OUTPUT ${fixture_files}
        COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/tests/make_fixture.py" --output "${fixture_root}"
        DEPENDS tests/make_fixture.py src/statue_definitions.h VERBATIM)
    add_custom_target(collector-statues-fixture DEPENDS ${fixture_files})
    add_dependencies(drop_tests collector-statues-fixture)
    add_test(NAME opening COMMAND opening_tests)
    add_test(NAME drops COMMAND drop_tests "${fixture_root}")
    add_test(NAME native_bridge COMMAND bridge_tests)
    add_test(NAME catalog COMMAND catalog_tests)
    add_test(NAME sound COMMAND sound_tests)
endif()
